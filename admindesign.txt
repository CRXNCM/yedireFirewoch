### 2. Sidebar Component
Create `src/components/admin/layout/Sidebar.jsx`:
```jsx
import React from 'react';
import { Link, useLocation } from 'react-router-dom';
import { 
  FaTachometerAlt, FaSchool, FaImages, FaUsers, 
  FaUserFriends, FaDonate, FaQuoteRight, FaHandshake,
  FaBullhorn, FaBell, FaCog
} from 'react-icons/fa';

const Sidebar = () => {
  const location = useLocation();
  
  const navItems = [
    { to: '/admin', icon: <FaTachometerAlt />, label: 'Dashboard' },
    { to: '/admin/schools', icon: <FaSchool />, label: 'Schools' },
    { to: '/admin/gallery', icon: <FaImages />, label: 'Gallery' },
    { to: '/admin/banks', icon: <FaDonate />, label: 'Bank Accounts' },
    { to: '/admin/testimonials', icon: <FaQuoteRight />, label: 'Testimonials' },
    { to: '/admin/sponsors', icon: <FaHandshake />, label: 'Sponsors' },
    { to: '/admin/communities', icon: <FaUsers />, label: 'Communities' },
    { to: '/admin/volunteers', icon: <FaUserFriends />, label: 'Volunteers' },
    { to: '/admin/social-media', icon: <FaBullhorn />, label: 'Social Media' },
    { to: '/admin/alerts', icon: <FaBell />, label: 'Alerts' },
    { to: '/admin/settings', icon: <FaCog />, label: 'Settings' },
  ];

  return (
    <div className="w-64 bg-white shadow-lg">
      <div className="p-4 border-b border-gray-200">
        <h1 className="text-xl font-bold text-primary">YEDIRE FIREWOCH</h1>
      </div>
      <nav className="p-4">
        <ul className="space-y-2">
          {navItems.map((item) => (
            <li key={item.to}>
              <Link
                to={item.to}
                className={`flex items-center p-3 rounded-lg transition-colors ${
                  location.pathname === item.to
                    ? 'bg-primary text-white'
                    : 'text-gray-600 hover:bg-gray-100'
                }`}
              >
                <span className="mr-3">{item.icon}</span>
                <span>{item.label}</span>
              </Link>
            </li>
          ))}
        </ul>
      </nav>
    </div>
  );
};

export default Sidebar;
```

## Component Library

### 1. Button Component
Create `src/components/admin/ui/Button.jsx`:
```jsx
import React from 'react';
import classNames from 'classnames';

const Button = ({
  children,
  variant = 'primary',
  size = 'md',
  className = '',
  ...props
}) => {
  const baseStyles = 'inline-flex items-center justify-center rounded-lg font-medium focus:outline-none focus:ring-2 focus:ring-offset-2 transition-colors';
  
  const variants = {
    primary: 'bg-primary text-white hover:bg-primary-dark focus:ring-primary',
    secondary: 'bg-secondary text-white hover:bg-secondary-dark focus:ring-secondary',
    outline: 'border border-gray-300 text-gray-700 bg-white hover:bg-gray-50 focus:ring-primary',
    danger: 'bg-red-600 text-white hover:bg-red-700 focus:ring-red-500',
    ghost: 'text-gray-600 hover:bg-gray-100 focus:ring-gray-200',
  };

  const sizes = {
    sm: 'px-3 py-1.5 text-sm',
    md: 'px-4 py-2 text-base',
    lg: 'px-6 py-3 text-lg',
  };

  return (
    <button
      className={classNames(
        baseStyles,
        variants[variant],
        sizes[size],
        className
      )}
      {...props}
    >
      {children}
    </button>
  );
};

export default Button;
```

### 2. Card Component
Create `src/components/admin/ui/Card.jsx`:
```jsx
import React from 'react';

const Card = ({ children, className = '', title, actions }) => {
  return (
    <div className={`bg-white rounded-lg shadow ${className}`}>
      {(title || actions) && (
        <div className="px-6 py-4 border-b border-gray-200 flex items-center justify-between">
          <h3 className="text-lg font-medium text-gray-900">{title}</h3>
          {actions && <div>{actions}</div>}
        </div>
      )}
      <div className="p-6">{children}</div>
    </div>
  );
};

export default Card;
```

## Implementation Notes

   - Uses Tailwind's responsive prefixes (sm:, md:, lg:, xl:)
   - Mobile-first approach with stacked layouts on small screens

2. **Accessibility**:
   - Proper ARIA labels
   - Keyboard navigation support
   - Sufficient color contrast

3. **Performance**:
   - Optimized with PurgeCSS in production
   - Lazy-loaded components where appropriate

4. **Customization**:
   - Theme can be easily modified via `tailwind.config.js`
   - Component variants for consistent styling

## Overview
Design a modern, clean, and intuitive admin dashboard for Yedire Firewoch, an educational non-profit organization. The dashboard should be professional, accessible, and optimized for ease of use.

## Color Scheme
- Primary: Deep blue (#1a365d) - Represents trust and professionalism
- Secondary: Gold (#d4af37) - Represents quality and excellence
- Accent: Teal (#2c7a7b) - For important actions and highlights
- Background: Light gray (#f7fafc) - For the main background
- Text: Dark gray (#2d3748) for primary text, medium gray (#718096) for secondary text
- Success: Green (#38a169)
- Warning: Yellow (#d69e2e)
- Danger: Red (#e53e3e)

## Typography
- Primary Font: 'Inter' (Sans-serif)
- Secondary Font: 'Roboto' (Sans-serif)
- Base Font Size: 16px
- Headings: 
  - H1: 2.25rem (36px)
  - H2: 1.875rem (30px)
  - H3: 1.5rem (24px)
  - H4: 1.25rem (20px)
